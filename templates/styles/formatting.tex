% Formatting definitions

% Color definitions
\definecolor{primarycolor}{RGB}{52, 73, 94}
\definecolor{secondarycolor}{RGB}{149, 165, 166}

% XeLaTeX Font Configuration with Linux-compatible fonts
% Using Liberation fonts for cross-platform compatibility
\setmainfont[
    Ligatures=TeX,
    Numbers=OldStyle
]{Liberation Serif}

% Alternative serif font family
\newfontfamily\timesroman[
    Ligatures=TeX,
    Scale=1.0
]{Liberation Serif}

% Sans-serif font using Liberation Sans
\setsansfont[
    Ligatures=TeX,
    Scale=0.9
]{Liberation Sans}

% Monospace font using Liberation Mono
\setmonofont[
    Scale=0.85,
    FakeStretch=0.9
]{Liberation Mono}

% Hyperlink setup
\hypersetup{
    colorlinks=true,
    linkcolor=primarycolor,
    urlcolor=primarycolor
}
