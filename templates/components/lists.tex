% List components for BRAT Annotation Guide Template
% Reusable list environments and formatting

% Checklist environment
\newlist{checklist}{itemize}{2}
\setlist[checklist]{label=$\square$}

% Custom checklist command
\newcommand{\checklistitem}[1]{\item #1}
\newcommand{\checkeditem}[1]{\item[$\boxtimes$] #1}

% Entity type list
\newenvironment{entitylist}
{%
    \begin{itemize}[label=\textcolor{primarycolor}{\textbf{E}}, leftmargin=1.5em]
}{%
    \end{itemize}
}

% Relation type list  
\newenvironment{relationlist}
{%
    \begin{itemize}[label=\textcolor{secondarycolor}{\textbf{R}}, leftmargin=1.5em]
}{%
    \end{itemize}
}

% Guideline steps list
\newenvironment{steplist}
{%
    \begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=2em]
}{%
    \end{enumerate}
}

% Requirements list
\newenvironment{reqlist}
{%
    \begin{itemize}[label=\textcolor{primarycolor}{\textbullet}, leftmargin=1.5em]
}{%
    \end{itemize}
}

% Examples list with special formatting
\newenvironment{examplelist}
{%
    \begin{itemize}[label=\textcolor{green!75!black}{\textbf{Ex:}}, leftmargin=2em]
}{%
    \end{itemize}
}

% Two-column comparison commands (simplified approach)
\newcommand{\begincompare}[2]{%
    \noindent
    \begin{minipage}{0.48\textwidth}
    \textbf{#1:}
    \begin{itemize}[label=\textcolor{primarycolor}{\textbullet}]
}
\newcommand{\switchcompare}[1]{%
    \end{itemize}
    \end{minipage}\hfill
    \begin{minipage}{0.48\textwidth}
    \textbf{#1:}
    \begin{itemize}[label=\textcolor{secondarycolor}{\textbullet}]
}
\newcommand{\finishcompare}{%
    \end{itemize}
    \end{minipage}
}
