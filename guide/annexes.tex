\section{Annexes}

\subsection{Glossaire}

\begin{description}
    \item[Annotation] Processus d'enrichissement d'un texte avec des informations sémantiques, syntaxiques ou pragmatiques.
    
    \item[Bactériémie] Présence de bactéries dans le sang, infection grave avec mortalité élevée nécessitant une prise en charge rapide.
    
    \item[BERT] Bidirectional Encoder Representations from Transformers - Architecture de réseau de neurones pour le traitement du langage naturel.
    
    \item[BGN] Bacille Gram Négatif - Catégorie de bactéries caractérisées par leur coloration en microbiologie.
    
    \item[BLSE] Bêta-Lactamase à Spectre Étendu - Enzyme produite par certaines bactéries conférant une résistance aux antibiotiques bêta-lactamines.
    
    \item[BRAT] Brat Rapid Annotation Tool - Outil web collaboratif pour l'annotation de textes.
    
    \item[CIM-10] Classification internationale des maladies, 10e révision - Système de codage standardisé des diagnostics médicaux.
    
    \item[CRH] Compte-Rendu d'Hospitalisation - Document médical récapitulatif du séjour hospitalier d'un patient.
    
    \item[Entité nommée] Élément textuel spécifique identifié et classifié (personne, lieu, organisation, etc.).
    
    \item[EPC] Entérobactérie Productrice de Carbapénémase - Bactérie résistante aux carbapénèmes.
    
    \item[ERV] Entérocoque Résistant à la Vancomycine - Souche d'entérocoque avec résistance acquise.
    
    \item[F1 score] Mesure harmonique entre la précision et le rappel, utilisée pour évaluer les performances des modèles de TAL.
    
    \item[Gold Standard] Corpus de référence annoté avec la plus haute qualité possible, utilisé pour l'évaluation.
    
    \item[HACEK] Groupe de bactéries (Haemophilus, Aggregatibacter, Cardiobacterium, Eikenella, Kingella) causes rares d'endocardite.
    
    \item[Inter-annotateur (accord)] Mesure de cohérence entre les annotations produites par différents annotateurs.
    
    \item[Kappa (coefficient)] Mesure statistique de l'accord inter-annotateurs qui prend en compte l'accord dû au hasard.
    
    \item[LLM] Large Language Model - Modèle de langage de grande taille basé sur l'apprentissage profond.
    
    \item[ORL] Oto-Rhino-Laryngologie - Spécialité médicale concernant l'oreille, le nez et la gorge.
    
    \item[PAC] Port-à-Cath - Chambre implantable pour administration intraveineuse.
    
    \item[PICC-line] Peripherally Inserted Central Catheter - Cathéter central inséré par voie périphérique.
    
    \item[PLP2a] Protéine Liant la Pénicilline 2a - Marqueur de résistance à la méthicilline chez S. aureus.
    
    \item[PNA] Pyélonéphrite Aiguë - Infection bactérienne aiguë du rein.
    
    \item[Relation] Lien sémantique entre deux ou plusieurs entités annotées.
    
    \item[SARM] Staphylococcus Aureus Résistant à la Méthicilline.
    
    \item[SASM] Staphylococcus Aureus Sensible à la Méthicilline.
    
    \item[Schéma d'annotation] Ensemble de règles et de types définissant comment annoter un corpus spécifique.
    
    \item[Site primaire] Porte d'entrée initiale de l'infection bactérienne dans l'organisme (ex: urines, peau, catheter).
    
    \item[Site secondaire] Localisation de complications ou d'extensions de l'infection bactérienne à partir du site primaire.
    
    \item[SNC] Système Nerveux Central - Ensemble formé par l'encéphale et la moelle épinière.
    
    \item[Standoff annotation] Format d'annotation où les annotations sont stockées séparément du texte original.
    
    \item[TAL] Traitement Automatique du Langage - Discipline informatique visant à automatiser l'analyse et le traitement des langues naturelles.
    
    \item[VRE] Vancomycin-Resistant Enterococcus - Voir ERV.
\end{description}

Person

Location

Money

Lives_in    Arg1:Person, Arg2:Location
Works_for   Arg1:Person, Arg2:Organization
Located_in  Arg1:Organization, Arg2:Location

[events]
Meeting     Theme:Person, Location:Location, Time:Date

[attributes]
Gender      Arg:Person, Value:Male|Female
Certainty   Arg:<ENTITY>, Value:High|Medium|Low
\end{verbatim}

\subsection{Métriques de qualité}

\subsubsection{Calcul du coefficient Kappa}
\[
\kappa = \frac{P(A) - P(E)}{1 - P(E)}
\]
où :
\begin{itemize}
    \item $P(A)$ : Proportion d'accord observé
    \item $P(E)$ : Proportion d'accord attendu par le hasard
\end{itemize}

\subsubsection{Interprétation des valeurs Kappa}
\begin{itemize}
    \item $\kappa < 0.20$ : Accord faible
    \item $0.20 \leq \kappa < 0.40$ : Accord correct
    \item $0.40 \leq \kappa < 0.60$ : Accord modéré
    \item $0.60 \leq \kappa < 0.80$ : Accord substantiel
    \item $\kappa \geq 0.80$ : Accord quasi parfait
\end{itemize}
